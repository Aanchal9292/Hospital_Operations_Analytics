-- =============================================================
-- Hospital Operations Analytics — DAX Measures
-- Paste into a dedicated "_Measures" table, or onto admissions/billing
-- =============================================================

Total Revenue =
SUM(billing[Total_Amount])

Total Patients =
DISTINCTCOUNT(patients[Patient_ID])

Total Admissions =
COUNTROWS(admissions)

Total Discharges =
CALCULATE(COUNTROWS(admissions), NOT ISBLANK(admissions[Discharge_Date]))

Average Revenue Per Patient =
DIVIDE([Total Revenue], [Total Patients])

Average Length of Stay =
AVERAGEX(
    FILTER(admissions, NOT ISBLANK(admissions[Discharge_Date])),
    DATEDIFF(admissions[Admission_Date], admissions[Discharge_Date], DAY)
)

Average Waiting Time =
AVERAGE(appointments[Waiting_Time])

Appointment Completion Rate =
DIVIDE(
    CALCULATE(COUNTROWS(appointments), appointments[Status] = "Completed"),
    COUNTROWS(appointments)
)

Cancellation Rate =
DIVIDE(
    CALCULATE(COUNTROWS(appointments), appointments[Status] = "Cancelled"),
    COUNTROWS(appointments)
)

No-Show Rate =
DIVIDE(
    CALCULATE(COUNTROWS(appointments), appointments[Status] = "No-show"),
    COUNTROWS(appointments)
)

Occupied Beds =
CALCULATE(COUNTROWS(beds), beds[Availability_Status] = "Occupied")

Available Beds =
CALCULATE(COUNTROWS(beds), beds[Availability_Status] = "Available")

Bed Occupancy Rate =
DIVIDE([Occupied Beds], COUNTROWS(beds))

Insurance Share % =
DIVIDE(SUM(billing[Insurance_Amount]), [Total Revenue])

Outstanding Payments =
CALCULATE(SUM(billing[Patient_Payment]), billing[Payment_Status] <> "Paid")
